async function initGoldParticles() { if ( particlesInitialized || prefersReducedMotion() || typeof window.tsParticles?.load !== 'function' ) return; const container = $('goldParticles'); if (!container) return; particlesInitialized = true; try { particlesContainer = await window.tsParticles.load('goldParticles', { fpsLimit: 30, detectRetina: true, particles: { number: { value: 34, density: { enable: true, area: 900 } }, color: { value: ['#d4af37', '#a2825e', '#e6c575'] }, shape: { type: 'circle' }, opacity: { value: { min: 0.12, max: 0.38 } }, size: { value: { min: 1, max: 2 } }, move: { enable: true, speed: 0.35, direction: 'none', outModes: { default: 'out' } }, links: { enable: false } }, interactivity: { events: { onHover: { enable: false }, onClick: { enable: false } } } }); } catch (error) { particlesInitialized = false; console.warn( 'Gold particle background could not be initialized:', error ); } } function attachTiltToCards() { if ( prefersReducedMotion() || !hasFinePointer() || !window.VanillaTilt?.init ) return; const cards = Array.from( document.querySelectorAll( '.product-card:not([data-tilt-active="1"])' ) ); if (!cards.length) return; window.VanillaTilt.init(cards, { max: 3, speed: 500, scale: 1.01, glare: false, perspective: 1000 }); cards.forEach(card => card.setAttribute('data-tilt-active', '1') ); } function initVisualLibraries() { if (!aosInitialized && window.AOS) { window.AOS.init({ once: true, offset: 30, duration: 550 }); aosInitialized = true; } refreshIcons(); initGoldParticles(); attachTiltToCards(); }
